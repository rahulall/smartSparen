-- ========================
-- 1. USERS TABLE
-- ========================
CREATE TABLE smart_sparen.users (
    user_id SERIAL PRIMARY KEY,
    name VARCHAR(255),
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
	updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ========================
-- 2. ACCOUNTS TABLE
-- ========================
CREATE TABLE smart_sparen.accounts (
    account_id SERIAL PRIMARY KEY,
    user_id INTEGER NOT NULL REFERENCES smart_sparen.users(user_id) ON DELETE CASCADE,
    account_name VARCHAR(255),
    account_type VARCHAR(50),  -- e.g. checking, credit
    balance NUMERIC(12, 2) DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
	updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_accounts_user_id ON smart_sparen.accounts(user_id);

-- ========================
-- 3. CATEGORIES TABLE
-- ========================
CREATE TABLE smart_sparen.category (
    category_id SERIAL PRIMARY KEY,
    category_name VARCHAR(255) UNIQUE NOT NULL,
	created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
	updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ========================
-- 4. EXPENSES TABLE
-- ========================
CREATE TABLE smart_sparen.expenses (
    expense_id SERIAL PRIMARY KEY,
    user_id INTEGER NOT NULL REFERENCES smart_sparen.users(user_id) ON DELETE CASCADE,
    account_id INTEGER REFERENCES smart_sparen.accounts(account_id) ON DELETE SET NULL,
    category_id INTEGER REFERENCES smart_sparen.category(category_id) ON DELETE SET NULL,
    amount NUMERIC(12,2) NOT NULL,
    description VARCHAR(255),
	merchant VARCHAR(255),
    transaction_date DATE NOT NULL,
	currency VARCHAR(10),
    is_manual_tag BOOLEAN DEFAULT FALSE,
	created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
	updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_expenses_user_id ON smart_sparen.expenses(user_id);
CREATE INDEX idx_expenses_account_id ON smart_sparen.expenses(account_id);
CREATE INDEX idx_expenses_category_id ON smart_sparen.expenses(category_id);
CREATE INDEX idx_expenses_transaction_date ON smart_sparen.expenses(transaction_date);

-- ========================
-- 5. BUDGETS TABLE
-- ========================
CREATE TABLE smart_sparen.budgets (
    budget_id SERIAL PRIMARY KEY,
    user_id INTEGER NOT NULL REFERENCES smart_sparen.users(user_id) ON DELETE CASCADE,
    category_id INTEGER REFERENCES smart_sparen.category(category_id) ON DELETE SET NULL,
    budget_amount NUMERIC(12, 2) NOT NULL,
	period_type VARCHAR(20) CHECK (period_type IN ('weekly', 'monthly', 'yearly')),
    start_date DATE NOT NULL,
    end_date DATE,
	alert_threshold NUMERIC(5,2) DEFAULT 0.9,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
	updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_budgets_user_id ON smart_sparen.budgets(user_id);
CREATE INDEX idx_budgets_category_id ON smart_sparen.budgets(category_id);

-- ========================
-- 6. BUDGET_ALERTS TABLE
-- ========================
CREATE TABLE smart_sparen.budget_alerts (
    alert_id SERIAL PRIMARY KEY,
    user_id INTEGER NOT NULL REFERENCES smart_sparen.users(user_id) ON DELETE CASCADE,
    budget_id INTEGER REFERENCES smart_sparen.budgets(budget_id) ON DELETE SET NULL,
    alert_type VARCHAR(50), -- e.g. LIMIT_WARNING, UNUSUAL_TRANSACTION
    message TEXT,
	created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
	updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_budget_alerts_user_id ON smart_sparen.budget_alerts(user_id);
CREATE INDEX idx_budget_alerts_budget_id ON smart_sparen.budget_alerts(budget_id);

-- ========================
-- 7. INSIGHTS TABLE
-- ========================
CREATE TABLE smart_sparen.insights (
    insight_id SERIAL PRIMARY KEY,
    user_id INTEGER NOT NULL REFERENCES smart_sparen.users(user_id) ON DELETE CASCADE,
    insight_type VARCHAR(100),  -- e.g. WEEKLY_SUMMARY, OVERSPEND
    insight_data JSONB,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
	updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_insights_user_id ON smart_sparen.insights(user_id);

-- ========================
-- 8. DASHBOARD_VIEWS TABLE
-- ========================
CREATE TABLE smart_sparen.dashboard_views (
    view_id SERIAL PRIMARY KEY,
    user_id INTEGER NOT NULL REFERENCES smart_sparen.users(user_id) ON DELETE CASCADE,
    view_type VARCHAR(100),  -- e.g. MONTHLY, WEEKLY
    view_data JSONB,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
	updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_dashboard_views_user_id ON smart_sparen.dashboard_views(user_id);


CREATE INDEX idx_transaction_id ON smart_sparen.transaction(transaction_id);

CREATE TABLE smart_sparen.goal (
    goal_id SERIAL PRIMARY KEY,
    user_id INT NOT NULL,
    account_id INT NOT NULL,
    category_id INT NOT NULL,
    goal_name VARCHAR(255) NOT NULL,
    target_amount NUMERIC(15, 2),
    goal_start_date DATE,
    goal_end_date DATE,
    goal_status VARCHAR(255),                -- e.g., 'active', 'completed', 'paused'
    frequency_of_contribution VARCHAR(255), -- e.g., 'weekly', 'monthly'
    contribution_style VARCHAR(255),        -- e.g., 'fixed', 'percentage'
    day_of_week VARCHAR(255),               -- e.g., 'Monday', NULL if not applicable
    date_of_month INT,                     -- 1-31, NULL if not applicable
    contribution_amount NUMERIC(15, 2),
    round_to_next_euro NUMERIC(15, 2),    -- amount to round contribution up to next euro
    percentage_of_expense NUMERIC(5, 2),  -- percentage like 12.34
    category_of_expense VARCHAR(255),
    created_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

 CONSTRAINT fk_goal_user FOREIGN KEY (user_id)
        REFERENCES smart_sparen.users (user_id)
        ON DELETE CASCADE,

    CONSTRAINT fk_goal_account FOREIGN KEY (account_id)
        REFERENCES smart_sparen.accounts (account_id)
        ON DELETE CASCADE,

    CONSTRAINT fk_goal_category FOREIGN KEY (category_id)
        REFERENCES smart_sparen.category (category_id)
        ON DELETE CASCADE
);

-- Select 2 months data category wise
SELECT
    e.user_id,
    c.category_name,
    EXTRACT(YEAR FROM e.transaction_date) AS year,
    TO_CHAR(e.transaction_date, 'Month') AS month,
    SUM(e.amount) AS total_expense
FROM
    smart_sparen.expenses e
JOIN
    smart_sparen.category c ON e.category_id = c.category_id
WHERE
    e.transaction_date >= DATE_TRUNC('month', CURRENT_DATE) - INTERVAL '1 month'
    AND e.transaction_date < DATE_TRUNC('month', CURRENT_DATE) + INTERVAL '1 month'
GROUP BY
    e.user_id,
    c.category_name,
    year,
    month
ORDER BY
    e.user_id,
    year,
    month,
    c.category_name;
